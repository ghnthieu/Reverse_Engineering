// ============================================================================
// MODULE: UTF-8 DECODER
// Input : Chuỗi std::string thô (đọc từ file bằng ifstream binary)
// Output: std::vector<uint32_t> chứa danh sách các Unicode Codepoint (RAM)
// ============================================================================
std::vector<uint32_t> decode_utf8(const std::string& raw_bytes) {
    std::vector<uint32_t> codepoints;
    size_t i = 0;
    size_t len = raw_bytes.length();

    while (i < len) {
        uint8_t c1 = static_cast<uint8_t>(raw_bytes[i]);
        uint32_t codepoint = 0;

        // Case 1: ASCII (1 byte) - Format: 0xxxxxxx
        if ((c1 & 0x80) == 0x00) {
            codepoint = c1;
            i += 1;
        }
        // Case 2: 2 bytes sequence - Format: 110xxxxx 10xxxxxx
        else if ((c1 & 0xE0) == 0xC0 && i + 1 < len) {
            uint8_t c2 = static_cast<uint8_t>(raw_bytes[i + 1]);
            codepoint = ((c1 & 0x1F) << 6) | (c2 & 0x3F);
            i += 2;
        }
        // Case 3: 3 bytes sequence - Format: 1110xxxx 10xxxxxx 10xxxxxx (Chữ Hán/Ký tự Unicode phổ biến)
        else if ((c1 & 0xF0) == 0xE0 && i + 2 < len) {
            uint8_t c2 = static_cast<uint8_t>(raw_bytes[i + 1]);
            uint8_t c3 = static_cast<uint8_t>(raw_bytes[i + 2]);
            codepoint = ((c1 & 0x0F) << 12) | ((c2 & 0x3F) << 6) | (c3 & 0x3F);
            i += 3;
        }
        // Case 4: 4 bytes sequence - Format: 11110xxx 10xxxxxx 10xxxxxx 10xxxxxx (Emoji, ký tự đặc biệt)
        else if ((c1 & 0xF8) == 0xF0 && i + 3 < len) {
            uint8_t c2 = static_cast<uint8_t>(raw_bytes[i + 1]);
            uint8_t c3 = static_cast<uint8_t>(raw_bytes[i + 2]);
            uint8_t c4 = static_cast<uint8_t>(raw_bytes[i + 3]);
            codepoint = ((c1 & 0x07) << 18) | ((c2 & 0x3F) << 12) | ((c3 & 0x3F) << 6) | (c4 & 0x3F);
            i += 4;
        }
        // Bỏ qua byte lỗi/không đúng định dạng
        else {
            i += 1;
            continue;
        }

        codepoints.push_back(codepoint);
    }

    return codepoints;
}
